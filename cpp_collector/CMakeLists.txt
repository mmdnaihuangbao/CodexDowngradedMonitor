cmake_minimum_required(VERSION 3.20)
project(CodexNativeCollector LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The collector uses the Windows read-only process API.")
endif()
add_executable(collector_native src/main.cpp)
target_compile_features(collector_native PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(collector_native PRIVATE /W4 /permissive- /utf-8)
endif()
